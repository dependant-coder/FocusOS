  medium: '#eab308',
  low: '#22c55e',

        initial={false}

        transition={{ type: 'spring', stiffness: 500, damping: 30 }}

            className="relative z-10 w-3 h-3 text-white"

      {/* Burst particles on completion */}

// components/daily/DailyTodoItem.tsx
import { motion, useMotionValue, useTransform, AnimatePresence } from 'framer-motion'
import { RiCheckLine, RiDeleteBinLine } from 'react-icons/ri'

interface SwipeableItemProps {
  onComplete: () => void
  onDelete: () => void
  children: React.ReactNode
}

export function SwipeableItem({ onComplete, onDelete, children }: SwipeableItemProps) {
  const x = useMotionValue(0)
  const background = useTransform(
    x,
    [-80, -20, 0, 20, 80],
    ['#ef4444', '#fca5a5', '#ffffff', '#bbf7d0', '#22c55e']
  )
  const leftOpacity  = useTransform(x, [0, 60], [0, 1])
  const rightOpacity = useTransform(x, [-60, 0], [1, 0])

  function handleDragEnd(_: unknown, info: { offset: { x: number } }) {
    if (info.offset.x > 72)       onComplete()
    else if (info.offset.x < -72) onDelete()
  }

  return (
    <div className="relative overflow-hidden rounded-xl mb-2">
      {/* Background action layer */}
      <motion.div
        className="absolute inset-0 flex items-center justify-between px-5"
        style={{ background }}
      >
        <motion.div style={{ opacity: leftOpacity }}>
          <RiCheckLine className="text-white text-xl" />
        </motion.div>
        <motion.div style={{ opacity: rightOpacity }}>
          <RiDeleteBinLine className="text-white text-xl" />
        </motion.div>
      </motion.div>

      {/* Draggable card */}
      <motion.div
        drag="x"
        dragConstraints={{ left: -100, right: 100 }}
        dragElastic={0.1}
        onDragEnd={handleDragEnd}
        style={{ x }}
        className="relative bg-white dark:bg-gray-800 cursor-grab active:cursor-grabbing"
        whileTap={{ cursor: 'grabbing' }}
      >

// hooks/useBulkSelect.ts
import { useState, useCallback } from 'react'

export function useBulkSelect() {
  const [selected, setSelected] = useState<Set<string>>(new Set())
  const [active, setActive]     = useState(false)

  const toggle = useCallback((id: string) => {
    setSelected(prev => {
      const next = new Set(prev)
      if (next.has(id)) next.delete(id)
      else               next.add(id)
      return next
    })
  }, [])

  const clear = useCallback(() => {
    setSelected(new Set())
    setActive(false)
  }, [])

  return { selected, active, setActive, toggle, clear, count: selected.size }
}

// components/vault/SortableTaskList.tsx
import {
  DndContext, closestCenter, PointerSensor,
  useSensor, useSensors, DragEndEvent,
} from '@dnd-kit/core'
import {
  SortableContext, verticalListSortingStrategy,
  useSortable, arrayMove,
} from '@dnd-kit/sortable'
import { CSS } from '@dnd-kit/utilities'

function SortableTaskCard({ task }: { task: Task }) {
  const { attributes, listeners, setNodeRef, transform, transition, isDragging } =
    useSortable({ id: task.id })

  return (
    <div
      ref={setNodeRef}
      style={{ transform: CSS.Transform.toString(transform), transition }}
      className={isDragging ? 'opacity-50 scale-[1.02] shadow-xl z-10 relative' : ''}
      {...attributes}
      {...listeners}
    >
      <TaskCard task={task} />
    </div>
  )
}

export function SortableTaskList({ tasks, onReorder }: { tasks: Task[]; onReorder: (tasks: Task[]) => void }) {
  const sensors = useSensors(useSensor(PointerSensor, { activationConstraint: { distance: 8 } }))

  function handleDragEnd(event: DragEndEvent) {
    const { active, over } = event
    if (over && active.id !== over.id) {
      const oldIndex = tasks.findIndex(t => t.id === active.id)
      const newIndex = tasks.findIndex(t => t.id === over.id)
      onReorder(arrayMove(tasks, oldIndex, newIndex))
    }
  }

  return (
    <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={handleDragEnd}>
      <SortableContext items={tasks.map(t => t.id)} strategy={verticalListSortingStrategy}>
        {tasks.map(task => <SortableTaskCard key={task.id} task={task} />)}
      </SortableContext>
    </DndContext>
  )
}

#### ✅ Keyboard Shortcut Map (Desktop / Physical Keyboard users)
```tsx
// hooks/useKeyboardShortcuts.ts
import { useEffect } from 'react'
import { useNavigate } from 'react-router-dom'

export function useKeyboardShortcuts(onAddTask: () => void) {
  const navigate = useNavigate()

  useEffect(() => {
    function handler(e: KeyboardEvent) {
      // Skip if user is typing in an input
      if (['INPUT', 'TEXTAREA'].includes((e.target as HTMLElement).tagName)) return

      if (e.key === 'n' && !e.metaKey && !e.ctrlKey) onAddTask()       // n = new task
      if (e.key === '1') navigate('/')           // 1 = Home
      if (e.key === '2') navigate('/daily')      // 2 = Daily
      if (e.key === '3') navigate('/vault')      // 3 = Vault
      if (e.key === '4') navigate('/habits')     // 4 = Habits
      if (e.key === '?') {/* show shortcuts modal */}
    }
    window.addEventListener('keydown', handler)
    return () => window.removeEventListener('keydown', handler)
  }, [navigate, onAddTask])
}
```
